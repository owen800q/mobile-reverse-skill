如果只是查看dex代码可以用MT管理器或jadx等等很多工具打开，但是有时候又会有需求想要修改dex中的代码并重打包，很多时候只是想改个返回值，直接返回ture，false，return，或者固定值，这里使用前辈已经造好的轮子如下，是个010Editor模板：

<https://www.sweetscape.com/010editor/repository/templates/file_info.php?file=DEX.bt&type=0&sort>=

比如我想改个返回值，需要先找到对应的类，再找到对应的方法，再找到方法方法对应的Opcode，以下是列举一些比较常用的反编译后的smali代码

 ![image](https://bbs.kanxue.com/upload/attach/202512/769188_T3SNQHH25BANZMT.webp)

每个方法的前16个字节都是描述方法有几个寄存器，占用几个字节码等等，描述信息在如下注释

![image](https://bbs.kanxue.com/upload/attach/202512/769188_45S8UNUNA7RZ2P3.webp)

剩下的才是真正的方法的字节码

12 10 0F 00对应aaaaa方法

14 00 01 00 00 00 0F 00对应bbbbb方法

14 00 10 27 00 00 0F 00对应fffff方法

 ![image](https://bbs.kanxue.com/upload/attach/202512/769188_T4J6GUF6UN97TCC.webp)

比如这我想把bbbbb方法改为返回100,那只需要把bbbbb方法对应的字节码改为

14 00 64 00 00 00 0F 00

![image](https://bbs.kanxue.com/upload/attach/202512/769188_8D2WEVB9BNU8ZJX.webp)

修改后还需要修改以下checksum和signature

![image](https://bbs.kanxue.com/upload/attach/202512/769188_DU4M2G38MZD7K24.webp)

如果没有修改的话打包后是会报错的，jadx打开也会报错

下面是自动计算checksum和signature代码

```
package com.dealSO;

import java.nio.ByteBuffer;
import java.nio.ByteOrder;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.zip.Adler32;

public class Teststr
{

	public static void main(String[] args)
	{

		String dexPathString = "D:\\AndroidKiller_v1.3.1\\projects\\getSN\\Bin\\classes.dex";
		byte[] modifiedDex;

		try
		{

			modifiedDex = Files.readAllBytes(Paths.get(dexPathString));
			// 6. 验证新的checksum
			long newChecksum = readCurrentChecksum(modifiedDex);
			System.out.println("old checksum: 0x" + Long.toHexString(newChecksum));

			// 更新checksum和signature
			byte[] fixedDex = updateDexChecksumAndSignature(modifiedDex);

			long newChecksum1 = readCurrentChecksum(modifiedDex);
			System.out.println("newChecksum1:" + newChecksum1);
			System.out.println("New checksum: 0x" + Long.toHexString(newChecksum1));

			Files.write(Paths.get(dexPathString), fixedDex);
		} catch (Exception e)
		{
			// TODO Auto-generated catch block
			e.printStackTrace();
		}

		System.out.println("Checksum fixed successfully!");

	}

	/**
	 * 计算DEX文件的Adler32 checksum DEX文件的checksum是从第12字节（跳过magic）到文件末尾的Adler32值
	 */
	public static long calculateDexChecksum(byte[] dexData)
	{
		Adler32 adler32 = new Adler32();

		// 从第12个字节开始（跳过8字节magic和4字节checksum自身）
		// magic: 8字节
		// checksum: 4字节
		// 所以从第12字节开始
		adler32.update(dexData, 12, dexData.length - 12);

		return adler32.getValue() & 0xFFFFFFFFL;
	}

	/**
	 * 计算DEX文件的SHA-1签名（signature）
	 */
	public static byte[] calculateDexSignature(byte[] dexData)
	{
		try
		{
			MessageDigest md = MessageDigest.getInstance("SHA-1");

			// SHA-1签名是从第32字节（跳过magic、checksum、signature）到文件末尾
			// magic: 8字节
			// checksum: 4字节
			// signature: 20字节
			// 所以从第32字节开始
			md.update(dexData, 32, dexData.length - 32);

			return md.digest();
		} catch (NoSuchAlgorithmException e)
		{
			throw new RuntimeException("SHA-1算法不可用", e);
		}
	}

	/**
	 * 更新DEX文件的checksum
	 */
	public static byte[] updateDexChecksum(byte[] dexData)
	{
		// 计算新的checksum
		long newChecksum = calculateDexChecksum(dexData);

		// 将checksum写入字节数组（小端序）
		ByteBuffer buffer = ByteBuffer.wrap(dexData);
		buffer.order(ByteOrder.LITTLE_ENDIAN);

		// checksum在偏移量8处（magic之后）
		buffer.position(8);
		buffer.putInt((int) newChecksum);

		return dexData;
	}

	/**
	 * 同时更新checksum和signature
	 */
	public static byte[] updateDexChecksumAndSignature(byte[] dexData)
	{
		// 1. 先更新signature（因为它不影响checksum计算）
		byte[] newSignature = calculateDexSignature(dexData);

		// 将signature写入字节数组
		ByteBuffer buffer = ByteBuffer.wrap(dexData);
		buffer.order(ByteOrder.LITTLE_ENDIAN);

		// signature在偏移量12处（magic和checksum之后）
		buffer.position(12);
		buffer.put(newSignature);

		// 2. 然后更新checksum
		return updateDexChecksum(dexData);
	}

	/**
	 * 读取当前的checksum
	 */
	public static long readCurrentChecksum(byte[] dexData)
	{
		ByteBuffer buffer = ByteBuffer.wrap(dexData);
		buffer.order(ByteOrder.LITTLE_ENDIAN);

		buffer.position(8);
		return buffer.getInt() & 0xFFFFFFFFL;
	}

	/**
	 * 验证DEX文件的checksum
	 */
	public static boolean verifyDexChecksum(byte[] dexData)
	{
		long current = readCurrentChecksum(dexData);
		long calculated = calculateDexChecksum(dexData);

		return current == calculated;
	}

	/**
	 * 完整的DEX修改流程
	 */
	public static byte[] modifyDexAndFixChecksum(byte[] originalDexData, DEXModifier modifier)
	{
		// 1. 修改DEX字节码
		byte[] modifiedDex = modifier.modify(originalDexData);

		// 2. 更新checksum和signature
		return updateDexChecksumAndSignature(modifiedDex);
	}

	/**
	 * 接口用于修改DEX内容
	 */
	public interface DEXModifier
	{
		byte[] modify(byte[] dexData);
	}

	/**
	 * Convert hex string to byte[]
	 *
	 * @param hexString
	 *            the hex string
	 * @return byte[]
	 */
	public static byte[] hexStringToBytes(String hexString)
	{
		if (hexString == null || hexString.equals(""))
		{
			return null;
		}
		hexString = hexString.toUpperCase();
		int length = hexString.length() / 2;
		char[] hexChars = hexString.toCharArray();
		byte[] d = new byte[length];
		for (int i = 0; i < length; i++)
		{
			int pos = i * 2;
			d[i] = (byte) (charToByte(hexChars[pos]) << 4 | charToByte(hexChars[pos + 1]));
		}
		return d;
	}

	/**
	 * Convert char to byte
	 *
	 * @param c
	 *            char
	 * @return byte
	 */
	private static byte charToByte(char c)
	{
		return (byte) "0123456789ABCDEF".indexOf(c);
	}

}
```
