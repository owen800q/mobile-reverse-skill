## BCF去除

这里很简单直接快速过了不仔细说了。

首先第一件事情就是去找不透明谓词的表达式 尽可能把不同的组合的都找齐全

表达式1：

```c
((((_BYTE)dword_68068 - 1) * (_BYTE)dword_68068) & 1) == 0 || dword_6807C < 10
\```

依旧带入值进行计算 dword_68068设为2 带入可得

```c
((( 2- 1) * 2) & 1) == 0 || dword_6807C < 10
\```

化简得到 按位与 结果为0 0等于0 所以左边恒为真 接着就是逻辑或 左边为真 就得看右边了 dword_6807C不知道具体值

```c
(( 2 & 1) == 0 || dword_6807C < 10
\```

去看dword_6807C的交叉引用 看看有没有写入赋值的片段 如果没有 就是初始值为0

根据类型来排序 清一色的offset和read 一个write没有

![image](https://bbs.kanxue.com/upload/attach/202609/951483_HVKFKFEPYVH9QM6.png)

那么意味着这个表达式恒为真

```c

手工分析已经证明谓词恒真，那这一整坨（ADRP 到 CSEL 之前）算出来的东西全是死的。但不能全删，因为最后那条 `CSEL W20, W9, W8, LT` 是在给状态寄存器写值， 只需要把 `CSEL W20, W9, W8, LT` 换成 `MOV W20, W9`。前面那十几条计算指令一条都不用动

接下来就是找锚点，两段的开头都是：

```java
ADRP  X8, #off_61D30@PAGE
LDR   X8, [X8,#off_61D30@PAGEOFF]
\```

`off_61D30` 是 GOT 表项，链接期就定死了，只有这个谓词会去读它 **。**

 ADRP 和 LDR 两条指令都会对 `off_61D30` 产生 xref，不过滤的话同一个谓词会被处理两遍。只认 ADRP，一个谓词就只进一次。

#### patch

只改 CSEL 那一条

```python
import idautils, idc, ida_bytes, ida_auto

for x in idautils.XrefsTo(idc.get_name_ea_simple("off_61D30")):
    ea = x.frm
    if idc.print_insn_mnem(ea).upper() != "ADRP":
        continue
    for _ in range(40):
        ea = idc.next_head(ea)
        mn = idc.print_insn_mnem(ea).upper()
        if mn == "CSEL":
            break
        if mn in ("B", "BL", "RET"):
            ea = None
            break
    if not ea or idc.print_operand(ea, 0).upper() != "W20":
        continue
    cond = idc.GetDisasm(ea).split(",")[-1].strip().upper()
    src = 1 if cond == "LT" else 2          # LT(TST形态)取Wn,NE(AND形态)取Wm
    rm = int(idc.print_operand(ea, src)[1:])
    ida_bytes.patch_dword(ea, 0x2A0003E0 | (rm << 16) | 20)
    ida_auto.plan_and_wait(ea, ea + 4)
    print(f"{ea:#x} -> MOV W20, W{rm}")
\```

有几个注意点 首先就是对W20的赋值方式有MOV、MOV+MOVK、CSEL

内部节点也分两个情况，一个是以B.EQ、B.NE的，在这里的被比较值w8就是其后继真实块的入口状态值

### 收集状态值 -> 真实块入口脚本

```java
import idc
import idautils
import ida_funcs

STATE_REG = "W20"
FUNC_ADDR = 0x5370      # JNI_OnLoad

def get_imm(ea, n):
    """取第 n 个操作数的立即数, 不是立即数返回 None"""
    if idc.get_operand_type(ea, n) == idc.o_imm:
        return idc.get_operand_value(ea, n) & 0xFFFFFFFF
    return None

def track(regs, ea):
    """维护 寄存器 -> 常量, 处理 MOV/MOVK 两种赋值形态"""
    mnem = idc.print_insn_mnem(ea)
    dst = idc.print_operand(ea, 0)

    if mnem == "MOV":
        imm = get_imm(ea, 1)
        if imm is not None:
            regs[dst] = imm                        # MOV Wd, #imm (含折叠的 MOVK)
        else:
            regs[dst] = regs.get(idc.print_operand(ea, 1))   # MOV Wd, Wm
    elif mnem == "MOVK":
        # 与 MOV 被业务指令隔开的情况, 例如 loc_6F94
        imm = get_imm(ea, 1)
        shift = 16 if "16" in idc.print_operand(ea, 1) else 0
        old = regs.get(dst)
        if old is not None and imm is not None:
            regs[dst] = old | (imm << shift)
    elif mnem in ("CMP", "CMN", "TST", "CCMP", "STR", "STUR", "STP", "B", "B.EQ", "B.NE"):
        pass                                        # 不改通用寄存器
    else:
        regs[dst] = None                            # 其他指令保守置未知

def collect():
    func = ida_funcs.get_func(FUNC_ADDR)
    state_map = {}
    regs = {}

    for ea in idautils.Heads(func.start_ea, func.end_ea):
        mnem = idc.print_insn_mnem(ea)

        # 找 CMP W20, Wm
        if mnem == "CMP" and idc.print_operand(ea, 0) == STATE_REG:
            src = idc.print_operand(ea, 1)
            state = regs.get(src)
            nxt = idc.next_head(ea, func.end_ea)
            cond = idc.print_insn_mnem(nxt)

            if state is None:
                print("[skip] 0x%X: %s 常量未知" % (ea, src))
            elif cond == "B.EQ":
                # 命中就跳走, 目标即真实块入口
                state_map[state] = idc.get_operand_value(nxt, 0)
            elif cond == "B.NE":
                # 不等回分发器, fall-through 才是命中
                state_map[state] = idc.next_head(nxt, func.end_ea)
            else:
                # GT/LE/LT/GE 是二分查找枢轴, 常量不是状态值
                print("[pivot] 0x%X: %s 0x%08X" % (ea, cond, state))

        track(regs, ea)

    return state_map

state_map = collect()

print("")
print("共 %d 个真实块" % len(set(state_map.values())))
for state in sorted(state_map):
    print("  0x%08X  ->  0x%X" % (state, state_map[state]))

### 真实块 -> 后继

```java
DISPATCHER = 0x53EC
TAIL_BLOCK = 0x53E0     # 尾块, CSEL W20,W8,W9,LT 的 LT 恒真
#最大步长
MAX_STEP   = 800

def solve_exit(entry, func_end):
    """
    从真实块入口顺序走到终结指令, 求出口状态值.
    返回 (类型, 数据, 终结地址)
      "one"     单后继, 数据 = 出口状态值
      "two"     双后继, 数据 = (条件, 值1, 值2, CSEL地址)
      "direct"  原生直连, 数据 = 目标地址
      "ret"     函数出口块
      "none"    算不出来, 交人工
    """
    regs = {}
    csel = None          # 真双后继时暂存
    term = None          # 终结地址, 以后打补丁用
    in_tail = False
    ea = entry

    for _ in range(MAX_STEP):
        #当前地址是主分发器就跳过
        if ea == DISPATCHER:            # 尾块 fallthrough 进分发器
            break

        mnem = idc.print_insn_mnem(ea)

        if mnem == "RET":
            return ("ret", None, ea)

        if mnem == "B":
            target = idc.get_operand_value(ea, 0)
            if target == TAIL_BLOCK:
                term = ea               # 补丁要打回源块这条 B
                in_tail = True
                ea = target
                continue
            if target == DISPATCHER:
                if term is None:
                    term = ea
                break
            return ("direct", target, ea)       # 真实块之间原生边

        # 栈保护检查: CMP X8, X9 + B.NE, 失败分支进 __stack_chk_fail, 直接跳过
        if mnem == "B.NE" and idc.print_insn_mnem(idc.prev_head(ea, entry)) == "CMP" \
                and idc.print_operand(ea - 4, 0).startswith("X"):
            ea = idc.next_head(ea, func_end)
            continue

        if mnem.startswith("B."):
            return ("none", None, None)

        if mnem == "CSEL" and idc.print_operand(ea, 0) == STATE_REG:
            src1 = idc.print_operand(ea, 1)
            src2 = idc.print_operand(ea, 2)
            if in_tail:
                regs[STATE_REG] = regs.get(src1)    # LT 恒真, 取第一个
            else:
                csel = (idc.print_operand(ea, 3), src1, src2, ea)
        else:
            track(regs, ea)

        ea = idc.next_head(ea, func_end)

    if csel is not None:
        cond, src1, src2, csel_ea = csel
        v1 = regs.get(src1)
        v2 = regs.get(src2)
        if v1 is None or v2 is None:
            return ("none", None, term)
        return ("two", (cond, v1, v2, csel_ea), term)

    value = regs.get(STATE_REG)
    if value is None or term is None:
        return ("none", None, term)
    return ("one", value, term)

#状态值->真实块地址
def build_succ(state_map):
    func = ida_funcs.get_func(FUNC_ADDR)

    succ = {}        # 真实块入口 -> (类型, 数据, 终结地址)
    manual = []

    for state in sorted(state_map):
        # 真实块入口地址
        entry = state_map[state]

        if entry in succ:
            continue                 # tail merging, 多个状态值共用一个块

        kind, data, term = solve_exit(entry, func.end_ea)

        if kind == "one":
            if data in state_map:
                succ[entry] = ("one", state_map[data], term)
            else:
                print("[人工] 0x%X 出口状态 0x%08X 查不到入口" % (entry, data))
                manual.append(entry)

        elif kind == "two":
            cond, v1, v2, csel_ea = data
            if v1 in state_map and v2 in state_map:
                succ[entry] = ("two", (cond, state_map[v1], state_map[v2], csel_ea), term)
            else:
                print("[人工] 0x%X 双后继状态查不到入口" % entry)
                manual.append(entry)

        elif kind == "direct":
            succ[entry] = ("direct", data, term)

        elif kind == "ret":
            succ[entry] = ("ret", None, term)

        else:
            print("[人工] 0x%X 出口状态算不出来" % entry)
            manual.append(entry)

    print("")
    print("真实块 %d 个, 解出 %d 个, 人工 %d 个"
          % (len(set(state_map.values())), len(succ), len(manual)))
    print("")
    for entry in sorted(succ):
        kind, data, term = succ[entry]
        if kind == "one":
            print("  0x%-6X -> 0x%-6X       (term 0x%X)" % (entry, data, term))
        elif kind == "two":
            cond, t1, t2, csel_ea = data
            print("  0x%-6X -> 0x%X / 0x%X  (%s, csel 0x%X)"
                  % (entry, t1, t2, cond, csel_ea))
        elif kind == "direct":
            print("  0x%-6X -> 0x%-6X       (原生直连)" % (entry, data))
        else:
            print("  0x%-6X -> RET" % entry)

    return succ

succ = build_succ(state_map)

```java
COND_CODE = {"EQ": 0x0, "NE": 0x1, "GE": 0xA, "LT": 0xB, "GT": 0xC, "LE": 0xD}

def enc_b(src, dst):
    offset = (dst - src) >> 2
    assert -(1 << 25) <= offset < (1 << 25)
    return 0x14000000 | (offset & 0x3FFFFFF)

def enc_bcond(src, dst, cond):
    offset = (dst - src) >> 2
    assert -(1 << 18) <= offset < (1 << 18)
    return 0x54000000 | ((offset & 0x7FFFF) << 5) | COND_CODE[cond]

def do_patch():
    count = 0
    for entry in sorted(succ):
        kind, data, term = succ[entry]

        if kind == "one":
            ida_bytes.patch_dword(term, enc_b(term, data))
            count += 1

        elif kind == "two":
            # CSEL Wd, Wn, Wm, cond -> 条件成立取 Wn, 即 t1
            cond, t1, t2, csel_ea = data
            ida_bytes.patch_dword(csel_ea, enc_bcond(csel_ea, t1, cond))
            ida_bytes.patch_dword(term, enc_b(term, t2))
            count += 2

        # direct / ret 不用动

    print("已 patch %d 处" % count)

do_patch()
