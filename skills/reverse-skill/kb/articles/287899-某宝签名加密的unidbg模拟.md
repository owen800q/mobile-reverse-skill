function findRealModuleByAddress(addr) {
    const modules = Process.enumerateModules();
    for (const m of modules) {
        if (addr.compare(m.base) >= 0 &&
            addr.compare(m.base.add(m.size)) < 0) {
            return m;
        }
    }
    // 检查匿名内存区域
    const range = Process.findRangeByAddress(addr);
    if (range && !range.file) {
        return {
            name: `[anon:${range.protection}]`,
            base: range.base,
            size: range.size
        };
    }
    return null;
}

有些方法是需要获取到app的数据的，可以先frida hook对应的方法，打印出入参和结果，模拟返回就可以。 对于方法的返回值类型是Object的，要特别注意，要返回对应的真实Class类型，不然抱的异常是没什么提示的，完全看不出来是什么原因。

\```
android.app.ActivityThread$ActivityClientRecord
\```

但是我返回了Object  就会出现奇奇怪怪的问题，可能不是这个，也可能是取Field 找不到字段之类的。
还有些是接口问题，如果本来有个类是实现了某个接口，但是我们在模拟Class时，没有给这个Class指定对应的接口，就会出现没有明确提示的错误，找不到方法或找不到MethodId之类的。所以在补环境的时候，要特别小心，如果找不到对应的方法，可以通过报错的methodId对应的hash码找找日志，可能在上面的日志中就有，但是没有指定继承关系，导致找不到。
