## 原理概述

QBDI 实现了一个自己的 ir, 然后将要插桩的二进制指令与插桩代码统一翻译成自己的 ir, 然后借助 llvm 的组件转成对应架构二进制指令, 天才的构思. 接下来让我们一点点揭开它神秘的面纱.

__qbdi_runCodeBlock:
    ; 检查 rsi 寄存器的第2位是否置位
    ; rsi 存储标志位：bit1=1 表示需要保存 FPU/SSE 状态
    test rsi, 2
    jz _skip_save_fpu                           ; 若未置位，跳过 FPU 保存流程

    ; ▼ 保存FPU/SSE状态到栈空间 ▼
    sub rsp, 8                                  ; 分配8字节栈空间（4字节 MXCSR + 2字节 FPU 控制字）
    stmxcsr [rsp]                               ; 保存SSE控制状态寄存器MXCSR到栈顶
    fnstcw [rsp+4]                              ; 保存 x87 FPU 控制字到栈顶 +4 位置

_skip_save_fpu:
    ; ▼ 保存调用者保留寄存器▼
    ; 遵循System V AMD64 ABI：rbx, rbp, r12-r15需由被调函数保存
    push r15
    push r14
    push r13
    push r12
    push rbp
    push rsi                                      ; 保存原始标志位（后续需用）
    push rbx

    ; ▼ 调用目标函数（通过寄存器传递的函数指针）▼
    ; rdi 存储目标函数地址（首个参数寄存器）
    call rdi

    ; ▼ 恢复调用者保留寄存器（逆序弹出）▼
    pop rbx
    pop rsi                                       ; 恢复原始标志位到rsi
    pop rbp
    pop r12
    pop r13
    pop r14
    pop r15

    ; ▼ 检查是否需要恢复 FPU 状态 ▼
    test rsi, 2
    jz _skip_restore_fpu

    ; ▼ 恢复 FPU/SSE 状态 ▼
    fninit                                        ; 初始化 FPU（重置状态）
    fldcw [rsp+4]                                 ; 从栈中恢复 FPU 控制字
    ldmxcsr [rsp]                                 ; 恢复 MXCSR 寄存器
    add rsp, 8                                    ; 释放栈空间

_skip_restore_fpu:
    cld                                           ; 清除方向标志（确保字符串操作正向增长）
